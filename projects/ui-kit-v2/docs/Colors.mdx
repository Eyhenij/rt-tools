export const Scale = ({ prefix, name, note }) => {
    const tokens = readTokens(prefix);
    if (!tokens.length) {
        return <p style={s.empty}>Ряда «{prefix}» в подключённых стилях нет.</p>;
    }
    return (
        <div style={s.scaleGrid}>
            <div>
                <div style={s.scaleTitle}>{name}</div>
                <div style={s.scaleNote}>{note}</div>
            </div>
            <div>
                <div style={s.strip}>
                    {tokens.map((token) => (
                        <div key={token.name} style={s.stripCell(`var(${token.name})`)} />
                    ))}
                </div>
                <div style={s.stripLegend}>
                    {tokens.map((token) => (
                        <div key={token.name} style={s.stripLabel} title={`${token.name} ${token.value}`}>
                            {token.name.replace(prefix, '')}
                            {tokens.length <= 12 ? (
                                <>
                                    <br />
                                    {token.value}
                                </>
                            ) : null}
                        </div>
                    ))}
                </div>
            </div>
        </div>
    );
};

export const THRESHOLD = 4.5;

export const parseRgb = (text) => {
    const parts = (text.match(/-?[\d.]+/g) ?? []).map(Number);
    return { r: parts[0] ?? 0, g: parts[1] ?? 0, b: parts[2] ?? 0, a: parts[3] ?? 1 };
};

export const over = (front, back) => ({
    r: front.r * front.a + back.r * (1 - front.a),
    g: front.g * front.a + back.g * (1 - front.a),
    b: front.b * front.a + back.b * (1 - front.a),
    a: 1,
});

export const luminance = ({ r, g, b }) => {
    const channel = (value) => {
        const part = value / 255;
        return part <= 0.03928 ? part / 12.92 : ((part + 0.055) / 1.055) ** 2.4;
    };
    return 0.2126 * channel(r) + 0.7152 * channel(g) + 0.0722 * channel(b);
};

export const ratioOf = (ink, backdrop) => {
    const [bright, dim] = [luminance(ink), luminance(backdrop)].sort((left, right) => right - left);
    return (bright + 0.05) / (dim + 0.05);
};

накладки разрешает сам браузер, и число не расходится с экраном. Половины тем берутся теми

export const measure = () => {
    if (typeof document === 'undefined') {
        return [];
    }
    const stage = document.createElement('div');
    stage.style.cssText = 'position:absolute;left:-9999px;top:0;visibility:hidden';
    document.body.appendChild(stage);
    const rows = PAIRS.map(([text, bg, where]) => ({ text, bg, where, light: null, dark: null }));
    for (const [theme, className] of [
        ['light', 'app-story-themes__pane'],
        ['dark', 'app-story-themes__pane app-story-themes__pane--dark'],
    ]) {
        const pane = document.createElement('div');
        pane.className = className;
        stage.appendChild(pane);
        const pageProbe = document.createElement('div');
        pageProbe.style.backgroundColor = 'var(--rt-color-bg-page)';
        pane.appendChild(pageProbe);
        const page = parseRgb(getComputedStyle(pageProbe).backgroundColor);
        for (const row of rows) {
            const probe = document.createElement('div');
            probe.style.color = `var(${row.text})`;
            probe.style.backgroundColor = `var(${row.bg})`;
            pane.appendChild(probe);
            const style = getComputedStyle(probe);
            const backdrop = parseRgb(style.backgroundColor);
            const ink = parseRgb(style.color);
            const solid = backdrop.a < 1 ? over(backdrop, page) : backdrop;
            row[theme] = ratioOf(ink.a < 1 ? over(ink, solid) : ink, solid);
        }
    }
    stage.remove();
    return rows;
};

export const Ratio = ({ value }) =>
    value === null ? (
        <span style={{ opacity: 0.6 }}>—</span>
    ) : (
        <span style={{ fontWeight: value < THRESHOLD ? 600 : 400, opacity: value < THRESHOLD ? 1 : 0.75 }}>
            {value.toFixed(2)}:1{value < THRESHOLD ? ' ✕' : ''}
        </span>
    );

export const Contrast = () => {
    const rows = measure();
    if (!rows.length) {
        return <p style={s.empty}>Замер идёт в браузере — здесь показывать нечего.</p>;
    }
    return (
        <table style={s.table}>
            <thead>
                <tr>
                    <th style={s.th}>где</th>
                    <th style={s.th}>текст</th>
                    <th style={s.th}>фон</th>
                    <th style={s.th}>светлая</th>
                    <th style={s.th}>тёмная</th>
                </tr>
            </thead>
            <tbody>
                {rows.map((row) => (
                    <tr key={`${row.text}|${row.bg}`}>
                        <td style={s.td}>{row.where}</td>
                        <td style={s.tdMono}>{row.text}</td>
                        <td style={s.tdMono}>{row.bg}</td>
                        <td style={s.td}>
                            <Ratio value={row.light} />
                        </td>
                        <td style={s.td}>
                            <Ratio value={row.dark} />
                        </td>
                    </tr>
                ))}
            </tbody>
        </table>
    );
};
