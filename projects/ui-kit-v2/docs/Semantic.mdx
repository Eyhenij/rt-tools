export const Preview = ({ prefix, limit = 6 }) => {
    const tokens = readTokens(prefix).slice(0, limit);
    if (!tokens.length) {
        return null;
    }
    return (
        <div style={s.panel}>
            {tokens.map((token) => (
                <div key={token.name} style={s.panelItem}>
                    <div style={s.panelPatch(`var(${token.name})`)} />
                    <span style={s.panelName}>{token.name}</span>
                </div>
            ))}
        </div>
    );
};

export const Group = ({ prefix }) => {
    const light = readTokens(prefix);
    if (!light.length) {
        return <p style={s.empty}>Назначений «{prefix}» в подключённых стилях нет.</p>;
    }
    const dark = new Map(readTokens(prefix, true).map((t) => [t.name, t.value]));
    return (
        <table style={s.table}>
            <thead>
                <tr>
                    <th style={s.th}>Token</th>
                    <th style={s.th}>Light</th>
                    <th style={s.th}>Dark</th>
                    <th style={s.th}>Применение</th>
                </tr>
            </thead>
            <tbody>
                {light.map((token) => (
                    <tr key={token.name}>
                        <td style={s.tdMono}>{token.name}</td>
                        <td style={s.tdMono}>
                            <span style={s.cellSwatch(`var(${token.name})`)} />
                            {token.value.replace('var(--rt-', '').replace(')', '')}
                        </td>
                        <td style={s.tdMono}>
                            {dark.has(token.name) ? (
                                <>
                                    <span style={s.cellSwatch(dark.get(token.name))} />
                                    {dark.get(token.name).replace('var(--rt-', '').replace(')', '')}
                                </>
                            ) : (
                                '— как в light'
                            )}
                        </td>
                        <td style={s.td}>{USAGE[token.name] ?? ''}</td>
                    </tr>
                ))}
            </tbody>
        </table>
    );
};
