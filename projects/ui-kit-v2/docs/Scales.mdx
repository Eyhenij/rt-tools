<Meta title="Foundation/Design Tokens/Spacing" />

export const Rows = ({ prefix, measure }) => {
    const tokens = readTokens(prefix);
    if (!tokens.length) {
        return <p style={s.empty}>Свойств «{prefix}» в подключённых стилях нет.</p>;
    }
    return (
        <div style={{ marginBlock: 16 }}>
            {tokens.map((token, index) => (
                <div key={token.name} style={s.scaleRow(index % 2 === 1)}>
                    <span style={s.scaleName}>{token.name}</span>
                    <span>{measure(token)}</span>
                    <span style={s.scaleValue}>{token.value}</span>
                </div>
            ))}
        </div>
    );
};

export const bar = (token) => (
    <span
        style={{
            display: 'block',
            height: 12,
            width: `var(${token.name})`,
            minWidth: 1,
            background: 'var(--rt-color-action-primary)',
            borderRadius: 2,
        }}
    />
);
export const corner = (token) => (
    <span
        style={{
            display: 'block',
            width: 52,
            height: 30,
            borderRadius: `var(${token.name})`,
            background: 'var(--rt-color-bg-surface-subtle-2)',
            border: '1px solid var(--rt-color-border-default)',
        }}
    />
);
export const sample = (token) => <span style={{ fontSize: `var(${token.name})` }}>Съешь ещё этих булок</span>;
export const heavy = (token) => <span style={{ fontWeight: `var(${token.name})` }}>Съешь ещё этих булок</span>;
export const shade = (token) => (
    <span
        style={{
            display: 'block',
            width: 60,
            height: 30,
            borderRadius: 6,
            background: 'var(--rt-color-bg-surface)',
            boxShadow: `var(${token.name})`,
        }}
    />
);
export const none = () => null;
